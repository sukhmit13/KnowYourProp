ADD TOD STATUS INCLUDING QUALIFYING BUS ROUTES

Chicago's TOD areas include properties within 0.25 miles of:
1. CTA rail stations
2. Metra stations  
3. High-frequency CTA bus routes (specified routes only)

STEP 2: UPDATE TOD CHECK LOGIC
═══════════════════════════════════════════════════════════════════════════

Expand TOD check to include qualifying bus routes:

const qualifyingBusRoutes = require('./data/tod_qualifying_bus_routes.json');

function checkTODStatus(transitStations) {
  // Check 1: Rail stations (CTA + Metra)
  const railStations = transitStations.filter(s => 
    s.type === 'CTA Rail' || s.type === 'Metra'
  );
  
  const closestRailStation = railStations[0];
  
  if (closestRailStation && closestRailStation.distance <= 0.25) {
    return {
      inTOD: true,
      todType: 'Rail Station',
      nearestStation: closestRailStation.name,
      stationType: closestRailStation.type,
      distance: closestRailStation.distance
    };
  }
  
  // Check 2: High-frequency bus routes
  const qualifyingBusStops = transitStations.filter(s => 
    s.type === 'CTA Bus' &&
    qualifyingBusRoutes.qualifying_routes.includes(s.route)
  );
  
  const closestQualifyingBus = qualifyingBusStops[0];
  
  if (closestQualifyingBus && closestQualifyingBus.distance <= 0.25) {
    return {
      inTOD: true,
      todType: 'High-Frequency Bus',
      nearestStation: closestQualifyingBus.name,
      busRoute: closestQualifyingBus.route,
      distance: closestQualifyingBus.distance
    };
  }
  
  // Not in TOD
  return {
    inTOD: false,
    closestRailStation: closestRailStation,
    closestQualifyingBus: closestQualifyingBus
  };
}

────────────────────────────────────

{IF inTOD AND todType === 'Rail Station'}
  ✅ Within TOD Area (Rail Transit)
  
  Nearest Rail Station:
  • {stationName} ({stationType})
  • Distance: {distance} miles
  
  TOD Benefits:
  • Reduced parking requirements
  • Streamlined development approvals
  • Transit-friendly development encouraged

{ELSE IF inTOD AND todType === 'High-Frequency Bus'}
  ✅ Within TOD Area (High-Frequency Bus)
  
  Nearest Qualifying Bus Stop:
  • Route {busRoute}: {stopName}
  • Distance: {distance} miles
  
  TOD Benefits:
  • Reduced parking requirements
  • Streamlined development approvals
  • Transit-friendly development encouraged

{ELSE}
  ❌ Not in TOD Area
  
  {IF closestRailStation OR closestQualifyingBus}
    Nearest qualifying transit:
    {IF closestRailStation AND distance < 0.5}
      • {railStationName} - {distance} miles (rail)
    {END IF}
    {IF closestQualifyingBus AND distance < 0.5}
      • Route {busRoute} - {distance} miles (high-freq bus)
    {END IF}
    
    Note: TOD area is within 0.25 miles of qualifying transit.
  {END IF}

{END IF}

- CTA rail stations (all lines)
- Metra stations (all lines)

2. Search Chicago Data Portal for:
   "high frequency bus routes"
   "transit priority corridors"

   URL: https://www.chicago.gov/city/en/depts/dcd/supp_info/tod.html

4. Look for routes with 10-minute or better peak service
