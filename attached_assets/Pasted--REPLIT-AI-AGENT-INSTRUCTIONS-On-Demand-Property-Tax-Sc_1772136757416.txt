## ARCHITECTURE

```
User requests report for address
    ↓
Get PIN from address (existing code)
    ↓
Call: get_tax_data(pin)
    ↓
Check cache: Does data/tax_cache/{pin}.json exist?
    ↓
    YES → Is it less than 90 days old?
            ↓
            YES → Return cached data (instant)
            NO → Scrape fresh data
    ↓
    NO → Scrape Cook County Treasurer website
    ↓
Cache the result as JSON file
    ↓
Return tax data to report
```

### Step 1: Create Cache Manager

Create file: `utils/cache.py`

```python
import json
import os
from datetime import datetime, timedelta

class TaxDataCache:
    """
    File-based cache for property tax data
    """
    
    def __init__(self, cache_dir='data/tax_cache'):
        self.cache_dir = cache_dir
        # Create cache directory if it doesn't exist
        os.makedirs(cache_dir, exist_ok=True)
    
    def get(self, pin):
        """
        Get cached tax data if it exists and is fresh (<90 days old)
        
        Args:
            pin: Property PIN (with or without dashes)
        
        Returns:
            dict: Cached tax data if fresh, None otherwise
        """
        pin_clean = pin.replace('-', '')
        cache_file = os.path.join(self.cache_dir, f"{pin_clean}.json")
        
        try:
            # Check if cache file exists
            if not os.path.exists(cache_file):
                print(f"[CACHE MISS] No cache file for PIN {pin}")
                return None
            
            # Read cache file
            with open(cache_file, 'r') as f:
                data = json.load(f)
            
            # Check if cache is still fresh (90 days)
            cached_date = datetime.fromisoformat(data['cached_at'])
            age_days = (datetime.now() - cached_date).days
            
            if age_days < 90:
                print(f"[CACHE HIT] PIN {pin} - Age: {age_days} days")
                return data['tax_data']
            else:
                print(f"[CACHE EXPIRED] PIN {pin} - Age: {age_days} days (>90)")
                return None
                
        except Exception as e:
            print(f"[CACHE ERROR] Error reading cache for PIN {pin}: {e}")
            return None
    
    def set(self, pin, tax_data):
        """
        Cache tax data to file
        
        Args:
            pin: Property PIN
            tax_data: Tax data dict to cache
        """
        pin_clean = pin.replace('-', '')
        cache_file = os.path.join(self.cache_dir, f"{pin_clean}.json")
        
        data = {
            'pin': pin,
            'cached_at': datetime.now().isoformat(),
            'tax_data': tax_data
        }
        
        try:
            with open(cache_file, 'w') as f:
                json.dump(data, f, indent=2)
            
            print(f"[CACHE SAVED] PIN {pin}")
            return True
            
        except Exception as e:
            print(f"[CACHE ERROR] Failed to save cache for PIN {pin}: {e}")
            return False
    
    def clear_expired(self):
        """
        Delete cache files older than 90 days
        """
        count = 0
        for filename in os.listdir(self.cache_dir):
            if filename.endswith('.json'):
                filepath = os.path.join(self.cache_dir, filename)
                
                try:
                    with open(filepath, 'r') as f:
                        data = json.load(f)
                    
                    cached_date = datetime.fromisoformat(data['cached_at'])
                    age_days = (datetime.now() - cached_date).days
                    
                    if age_days >= 90:
                        os.remove(filepath)
                        count += 1
                        print(f"[CACHE CLEANUP] Deleted expired cache: {filename}")
                
                except Exception as e:
                    print(f"[CACHE CLEANUP ERROR] {filename}: {e}")
        
        print(f"[CACHE CLEANUP] Removed {count} expired files")
        return count
```

### Step 2: Create Tax Scraper

Create file: `scrapers/tax_scraper.py`

```python
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.chrome.options import Options
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from bs4 import BeautifulSoup
import time
import re
from datetime import datetime

class CookCountyTaxScraper:
    """
    Scraper for Cook County Treasurer property tax data
    """
    
    def __init__(self):
        self.base_url = "https://www.cookcountytreasurer.com/pinsearch.aspx"
    
    def scrape_tax_data(self, pin):
        """
        Scrape property tax data from Cook County Treasurer
        
        Args:
            pin: Property PIN (with or without dashes)
        
        Returns:
            dict: Tax data including years, amounts, payment status
        """
        pin_clean = pin.replace('-', '')
        url = f"{self.base_url}?pin={pin_clean}"
        
        print(f"[SCRAPER] Scraping tax data for PIN {pin}...")
        print(f"[SCRAPER] URL: {url}")
        
        # Setup headless Chrome
        options = Options()
        options.add_argument('--headless')
        options.add_argument('--no-sandbox')
        options.add_argument('--disable-dev-shm-usage')
        options.add_argument('--disable-gpu')
        options.add_argument('--window-size=1920,1080')
        options.add_argument('user-agent=Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36')
        
        driver = None
        
        try:
            driver = webdriver.Chrome(options=options)
            driver.get(url)
            
            # Wait for page to load
            print("[SCRAPER] Waiting for page to load...")
            time.sleep(4)  # Give page time to render
            
            # Get page source
            html = driver.page_source
            soup = BeautifulSoup(html, 'html.parser')
            
            # Parse tax data
            tax_years = self._parse_tax_data(soup)
            
            if not tax_years:
                print("[SCRAPER] No tax data found - check if page structure changed")
                # Save HTML for debugging
                with open(f'debug_{pin_clean}.html', 'w') as f:
                    f.write(html)
                print(f"[SCRAPER] Saved page HTML to debug_{pin_clean}.html for inspection")
            
            # Calculate summary
            overall_status = self._determine_status(tax_years)
            total_outstanding = sum(year.get('balance', 0) for year in tax_years)
            
            result = {
                'pin': pin,
                'tax_years': tax_years,
                'overall_status': overall_status,
                'total_outstanding': total_outstanding,
                'scraped_at': datetime.now().isoformat(),
                'source_url': url
            }
            
            print(f"[SCRAPER] Successfully scraped {len(tax_years)} tax years")
            return result
            
        except Exception as e:
            print(f"[SCRAPER ERROR] {str(e)}")
            return {
                'pin': pin,
                'error': str(e),
                'status': 'Error - unable to retrieve tax data',
                'check_manually_url': url
            }
        
        finally:
            if driver:
                driver.quit()
    
    def _parse_tax_data(self, soup):
        """
        Parse tax data from HTML
        
        NOTE: You need to inspect the Cook County Treasurer page to get
        the correct selectors. This is a template - UPDATE based on actual HTML.
        """
        tax_years = []
        
        # METHOD 1: Try to find table with tax data
        # Look for tables that might contain tax information
        tables = soup.find_all('table')
        
        for table in tables:
            # Check if this looks like a tax table
            table_text = table.get_text()
            if any(keyword in table_text.lower() for keyword in ['tax year', 'total', 'billed', 'paid', 'balance']):
                
                rows = table.find_all('tr')
                
                for row in rows[1:]:  # Skip header row
                    cells = row.find_all(['td', 'th'])
                    
                    # Need at least 4 columns: year, billed, paid, balance
                    if len(cells) >= 4:
                        try:
                            year = cells[0].get_text().strip()
                            
                            # Only process if it looks like a year
                            if re.match(r'20\d{2}', year):
                                tax_years.append({
                                    'year': year,
                                    'total_billed': self._parse_currency(cells[1].get_text()),
                                    'amount_paid': self._parse_currency(cells[2].get_text()),
                                    'balance': self._parse_currency(cells[3].get_text()),
                                    'status': 'Paid' if self._parse_currency(cells[3].get_text()) <= 0 else 'Outstanding'
                                })
                        except Exception as e:
                            print(f"[SCRAPER] Error parsing row: {e}")
                            continue
        
        # METHOD 2: Try alternative selectors if Method 1 fails
        if not tax_years:
            # Try finding divs or other containers
            # UPDATE THESE SELECTORS based on actual page structure
            year_sections = soup.find_all('div', class_='tax-year')
            
            for section in year_sections:
                try:
                    # This is a placeholder - adjust based on actual HTML
                    year_elem = section.find('span', class_='year')
                    billed_elem = section.find('span', class_='billed')
                    paid_elem = section.find('span', class_='paid')
                    balance_elem = section.find('span', class_='balance')
                    
                    if all([year_elem, billed_elem, paid_elem, balance_elem]):
                        tax_years.append({
                            'year': year_elem.get_text().strip(),
                            'total_billed': self._parse_currency(billed_elem.get_text()),
                            'amount_paid': self._parse_currency(paid_elem.get_text()),
                            'balance': self._parse_currency(balance_elem.get_text()),
                            'status': 'Paid' if self._parse_currency(balance_elem.get_text()) <= 0 else 'Outstanding'
                        })
                except:
                    continue
        
        return tax_years
    
    def _parse_currency(self, text):
        """
        Convert currency text to float
        Examples: '$1,234.56' -> 1234.56, '(123.45)' -> -123.45
        """
        if not text:
            return 0.0
        
        # Remove whitespace
        text = text.strip()
        
        # Handle parentheses (negative numbers)
        is_negative = text.startswith('(') and text.endswith(')')
        
        # Remove currency symbols, commas, parentheses
        clean = re.sub(r'[$,\s()]', '', text)
        
        # Handle different negative signs
        clean = clean.replace('−', '-')
        
        try:
            value = float(clean)
            return -value if is_negative else value
        except ValueError:
            return 0.0
    
    def _determine_status(self, tax_years):
        """
        Determine overall tax payment status
        """
        if not tax_years:
            return 'Unknown'
        
        total_outstanding = sum(year.get('balance', 0) for year in tax_years)
        
        if total_outstanding <= 0:
            return 'Current'
        elif total_outstanding < 1000:
            return 'Minor Outstanding'
        else:
            return 'Delinquent'
```

### Step 3: Create Main Tax Data Fetcher

Create file: `scrapers/__init__.py` or add to existing module:

```python
from utils.cache import TaxDataCache
from scrapers.tax_scraper import CookCountyTaxScraper

class TaxDataFetcher:
    """
    Main interface for getting tax data
    Handles caching and on-demand scraping
    """
    
    def __init__(self):
        self.cache = TaxDataCache()
        self.scraper = CookCountyTaxScraper()
    
    def get_tax_data(self, pin):
        """
        Get tax data for a PIN - from cache if available, otherwise scrape
        
        Args:
            pin: Property PIN (with or without dashes)
        
        Returns:
            dict: Tax data including payment history and status
        """
        print(f"\n[TAX DATA] Fetching data for PIN {pin}")
        
        # Try cache first
        cached_data = self.cache.get(pin)
        
        if cached_data:
            print("[TAX DATA] Using cached data")
            return cached_data
        
        # Cache miss - scrape fresh data
        print("[TAX DATA] Cache miss - scraping fresh data...")
        scraped_data = self.scraper.scrape_tax_data(pin)
        
        # Cache the result (even if error, so we don't retry immediately)
        if scraped_data and 'error' not in scraped_data:
            self.cache.set(pin, scraped_data)
        
        return scraped_data
```

```python
# Add to tax_scraper.py

import time
from datetime import datetime

class RateLimiter:
    def __init__(self, requests_per_minute=20):
        self.max_requests = requests_per_minute
        self.requests = []
    
    def wait_if_needed(self):
        now = datetime.now()
        
        # Remove old requests
        self.requests = [r for r in self.requests if (now - r).seconds < 60]
        
        # Check limit
        if len(self.requests) >= self.max_requests:
            wait_time = 60 - (now - min(self.requests)).seconds
            if wait_time > 0:
                print(f"[RATE LIMIT] Waiting {wait_time}s...")
                time.sleep(wait_time)
        
        self.requests.append(now)
```

**What this does:**
1. User requests report
2. System checks if tax data is cached (<90 days old)
3. If cached → return instantly (free, fast)
