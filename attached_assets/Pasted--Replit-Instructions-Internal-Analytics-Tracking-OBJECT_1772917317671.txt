## STEP 1: CREATE DATABASE TABLE

Create a new migration file or add to your existing database schema:

```sql
-- Create analytics tracking table
CREATE TABLE IF NOT EXISTS search_analytics (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  
  -- Property info
  address TEXT NOT NULL,
  pin TEXT,
  neighborhood TEXT,
  zoning TEXT,
  
  -- Project details
  project_type TEXT,              -- residential, commercial, daycare, restaurant, etc.
  
  -- Property financials
  tax_assessed_value REAL,
  annual_tax_bill REAL,
  
  -- Search metadata
  searched_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  customer_email TEXT NOT NULL,
  customer_type TEXT,             -- 'single' or 'subscriber'
  report_id TEXT,
  
  -- Create indexes for fast queries
  FOREIGN KEY (report_id) REFERENCES reports(id)
);

CREATE INDEX IF NOT EXISTS idx_analytics_address ON search_analytics(address);
CREATE INDEX IF NOT EXISTS idx_analytics_project_type ON search_analytics(project_type);
CREATE INDEX IF NOT EXISTS idx_analytics_customer_type ON search_analytics(customer_type);
CREATE INDEX IF NOT EXISTS idx_analytics_searched_at ON search_analytics(searched_at);

-- Create view for quick summary stats
CREATE VIEW IF NOT EXISTS analytics_summary AS
SELECT 
  project_type,
  customer_type,
  COUNT(*) as total_searches,
  COUNT(DISTINCT address) as unique_properties,
  COUNT(DISTINCT customer_email) as unique_customers,
  AVG(tax_assessed_value) as avg_assessed_value,
  AVG(annual_tax_bill) as avg_tax_bill
FROM search_analytics
GROUP BY project_type, customer_type;
```

import sqlite3
from datetime import datetime

def track_search_analytics(address, customer_email, report_id, project_type=None):
    """
    Track search analytics internally
    
    Args:
        address: Property address
        customer_email: Customer's email
        report_id: Generated report ID
        project_type: Optional - residential, commercial, daycare, restaurant, etc.
    """
    try:
        # Get database connection
        conn = get_db_connection()  # Use your existing DB connection method
        cursor = conn.cursor()
        
        # Get property data (use your existing functions)
        property_data = get_property_data_from_cache_or_scrape(address)
        
        # Determine customer type
        cursor.execute("SELECT COUNT(*) FROM subscribers WHERE email = ?", (customer_email,))
        is_subscriber = cursor.fetchone()[0] > 0
        customer_type = 'subscriber' if is_subscriber else 'single'
        
        # Extract key property metrics
        pin = property_data.get('pin', '')
        neighborhood = property_data.get('neighborhood', '')
        zoning = property_data.get('zoning', '')
        assessed_value = property_data.get('tax_assessed_value', 0)
        annual_tax = property_data.get('annual_tax_bill', 0)
        
        # Insert analytics record
        cursor.execute("""
            INSERT INTO search_analytics 
            (address, pin, neighborhood, zoning, project_type, 
             tax_assessed_value, annual_tax_bill, 
             customer_email, customer_type, report_id)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        """, (
            address,
            pin,
            neighborhood,
            zoning,
            project_type,
            assessed_value,
            annual_tax,
            customer_email,
            customer_type,
            report_id
        ))
        
        conn.commit()
        
        # Check if we've hit 500 unique reports milestone
        check_milestone(cursor)
        
        conn.close()
        
    except Exception as e:
        print(f"Analytics tracking error: {e}")
        # Don't let analytics tracking break the main flow
        pass


def check_milestone(cursor):
    """
    Check if we've hit 500 unique reports and notify
    """
    cursor.execute("SELECT COUNT(DISTINCT report_id) as total FROM search_analytics")
    total_reports = cursor.fetchone()[0]
    
    # Check if we just crossed 500
    if total_reports == 500:
        send_milestone_notification(total_reports)
    
    # Also notify at other milestones (100, 250, 1000, etc.)
    milestones = [100, 250, 500, 750, 1000, 2500, 5000, 10000]
    if total_reports in milestones:
        send_milestone_notification(total_reports)

def get_quick_stats():
    """
    Get quick summary stats for milestone notification
    """
    conn = get_db_connection()
    cursor = conn.cursor()
    
    # Get breakdown by project type
    cursor.execute("""
        SELECT 
            project_type,
            COUNT(*) as count,
            ROUND(COUNT(*) * 100.0 / (SELECT COUNT(*) FROM search_analytics), 1) as pct
        FROM search_analytics
        WHERE project_type IS NOT NULL
        GROUP BY project_type
        ORDER BY count DESC
        LIMIT 5
    """)
    project_breakdown = cursor.fetchall()
    
    # Get breakdown by customer type
    cursor.execute("""
        SELECT 
            customer_type,
            COUNT(*) as count,
            ROUND(COUNT(*) * 100.0 / (SELECT COUNT(*) FROM search_analytics), 1) as pct
        FROM search_analytics
        GROUP BY customer_type
    """)
    customer_breakdown = cursor.fetchall()
    
    # Get top neighborhoods
    cursor.execute("""
        SELECT 
            neighborhood,
            COUNT(*) as count
        FROM search_analytics
        WHERE neighborhood IS NOT NULL AND neighborhood != ''
        GROUP BY neighborhood
        ORDER BY count DESC
        LIMIT 5
    """)
    neighborhoods = cursor.fetchall()
    
    conn.close()
    
    stats = "\nQUICK STATS:\n\n"
    
    stats += "By Customer Type:\n"
    for row in customer_breakdown:
        stats += f"  {row[0]}: {row[1]} ({row[2]}%)\n"
    
    stats += "\nTop Project Types:\n"
    for row in project_breakdown:
        stats += f"  {row[0]}: {row[1]} ({row[2]}%)\n"
    
    stats += "\nTop Neighborhoods:\n"
    for row in neighborhoods:
        stats += f"  {row[0]}: {row[1]} searches\n"
    
    return stats

from flask import render_template_string

@app.route('/admin/analytics')
def admin_analytics():
    """
    Simple admin view of analytics
    Password protect this route in production!
    """
    conn = get_db_connection()
    cursor = conn.cursor()
    
    # Total reports
    cursor.execute("SELECT COUNT(DISTINCT report_id) as total FROM search_analytics")
    total_reports = cursor.fetchone()[0]
    
    # Get summary data
    cursor.execute("SELECT * FROM analytics_summary ORDER BY total_searches DESC")
    summary_data = cursor.fetchall()
    
    # Recent searches
    cursor.execute("""
        SELECT address, project_type, customer_type, searched_at
        FROM search_analytics
        ORDER BY searched_at DESC
        LIMIT 20
    """)
    recent_searches = cursor.fetchall()
    
    # Top neighborhoods
    cursor.execute("""
        SELECT neighborhood, COUNT(*) as searches
        FROM search_analytics
        WHERE neighborhood IS NOT NULL AND neighborhood != ''

        ORDER BY searches DESC
        LIMIT 10

## OPTIONAL: EMAIL NOTIFICATION

If you want email notifications when hitting milestones, add this:

```python
import smtplib
from email.mime.text import MIMEText
import os

def send_email_notification(message):
    """
    Send email notification (optional)
    """
    try:
        # Set these as Replit Secrets
        smtp_host = os.getenv('SMTP_HOST', 'smtp.gmail.com')
        smtp_port = int(os.getenv('SMTP_PORT', '587'))
        smtp_user = os.getenv('SMTP_USER')
        smtp_pass = os.getenv('SMTP_PASS')
        notification_email = os.getenv('NOTIFICATION_EMAIL')
        
        if not all([smtp_user, smtp_pass, notification_email]):
            return  # Skip if not configured
        
        msg = MIMEText(message, 'plain')
        msg['Subject'] = '🎉 Analytics Milestone Reached!'
        msg['From'] = smtp_user
        msg['To'] = notification_email
        
        with smtplib.SMTP(smtp_host, smtp_port) as server:
            server.starttls()
            server.login(smtp_user, smtp_pass)
            server.send_message(msg)
            
        print("Email notification sent!")
        
    except Exception as e:
        print(f"Failed to send email notification: {e}")
```

Then add to Replit Secrets:
- `SMTP_HOST` = smtp.gmail.com
- `SMTP_PORT` = 587
- `SMTP_USER` = your-email@gmail.com
- `SMTP_PASS` = your-app-password
- `NOTIFICATION_EMAIL` = where-to-send-alerts@email.com
