Add a helper function to check if a location is within corridor bounds:

def _is_in_corridor_bounds(latitude, longitude, corridor_key):
    """Check if coordinates fall within corridor's geographic bounds."""
    corridor = CORRIDOR_KEYWORDS.get(corridor_key, {})
    if not all(k in corridor for k in ['lat_min', 'lat_max', 'lng_min', 'lng_max']):
        return False
    
    return (corridor['lat_min'] <= latitude <= corridor['lat_max'] and
            corridor['lng_min'] <= longitude <= corridor['lng_max'])

def _article_mentions_corridor(article, corridor_key):

Add a new helper function to detect corridors from both address AND coordinates:

def extract_corridor_from_address(address, latitude, longitude):
    """
    Determine which corridor (if any) a property is on.
    Uses BOTH address string matching AND geographic bounds verification.
    Returns dict with corridor info or None.
    """
    from news_monitor import CORRIDOR_KEYWORDS, _is_in_corridor_bounds
    
    address_lower = address.lower()
    
    for corridor_key, info in CORRIDOR_KEYWORDS.items():
        # Check if address string mentions the corridor
        for name in info['names']:
            if name in address_lower:
                # Verify the property is actually within corridor geographic bounds
                if _is_in_corridor_bounds(latitude, longitude, corridor_key):
                    return {
                        'corridor_key': corridor_key,
                        'corridor_name': name.title(),
                        'tier': info['tier'],
                        'description': info['description']
                    }
    
    return None
