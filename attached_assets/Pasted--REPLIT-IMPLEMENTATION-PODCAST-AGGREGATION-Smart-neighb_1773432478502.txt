## ============================================
## MAIN SCRAPING FUNCTION
## ============================================

def get_podcast_content_for_property(neighborhood, property_type='residential', days_back=60):
    """
    Main function: Get relevant podcast episodes for a property
    
    Parameters:
    - neighborhood: "Wicker Park", "Logan Square", etc.
    - property_type: 'residential', 'commercial', 'restaurant', etc.
    - days_back: How many days of episodes to check
    
    Returns:
    - Dictionary with relevant episodes by track
    """
    
    print(f"\n🎙️ Searching podcasts for {neighborhood}...")
    
    # Determine which feeds to check based on property type
    feeds_to_check = get_relevant_feeds(property_type)
    
    # Scrape all feeds
    all_episodes = []
    for feed_id in feeds_to_check:
        feed_info = PODCAST_FEEDS[feed_id]
        print(f"  → Checking {feed_info['name']}...")
        
        episodes = scrape_podcast_feed(
            feed_info['rss'],
            feed_info['name'],
            days_back
        )
        
        # Add metadata
        for ep in episodes:
            ep['feed_id'] = feed_id
            ep['category'] = feed_info['category']
            ep['track'] = feed_info['track']
        
        all_episodes.extend(episodes)
    
    print(f"  → Found {len(all_episodes)} total episodes")
    
    # Filter for neighborhood relevance
    relevant_episodes = filter_by_neighborhood(all_episodes, neighborhood)
    
    print(f"  → {len(relevant_episodes)} relevant to {neighborhood}")
    
    # Organize by track
    organized = {
        'hospitality': [],
        'real_estate': []
    }
    
    for ep in relevant_episodes:
        organized[ep['track']].append(ep)
    
    return organized


## ============================================
## FEED SELECTION BY PROPERTY TYPE
## ============================================

def get_relevant_feeds(property_type):
    """
    Determine which podcast feeds to check based on property type
    """
    # Always check Tier 1 (daily news)
    feeds = ['crains_gist', 'reset_wbez']
    
    # Add category-specific feeds
    if property_type in ['restaurant', 'bar', 'hospitality']:
        feeds.extend(['eater_digest', 'good_beer_hunting'])
    
    elif property_type in ['residential', 'condo', 'apartment']:
        feeds.extend(['chicago_tonight'])
    
    elif property_type in ['commercial', 'office', 'retail', 'mixed_use']:
        feeds.extend(['chicago_tonight'])
    
    else:
        # Default: check all
        feeds = list(PODCAST_FEEDS.keys())
    
    return feeds


## ============================================
## RSS FEED SCRAPER
## ============================================

def scrape_podcast_feed(rss_url, podcast_name, days_back=60):
    """
    Scrape podcast RSS feed
    """
    try:
        feed = feedparser.parse(rss_url)
        episodes = []
        cutoff_date = datetime.now() - timedelta(days=days_back)
        
        for entry in feed.entries:
            # Parse date
            if hasattr(entry, 'published_parsed'):
                try:
                    pub_date = datetime(*entry.published_parsed[:6])
                except:
                    continue
            else:
                continue
            
            # Skip if too old
            if pub_date < cutoff_date:
                continue
            
            # Extract episode info
            episode = {
                'title': entry.get('title', ''),
                'description': entry.get('summary', ''),
                'link': entry.get('link', ''),
                'published': pub_date.strftime('%B %d, %Y'),
                'pub_date_obj': pub_date,
                'podcast_name': podcast_name,
                'duration': entry.get('itunes_duration', 'Unknown')
            }
            
            episodes.append(episode)
        
        return episodes
    
    except Exception as e:
        print(f"    Error scraping {podcast_name}: {e}")
        return []


## ============================================
## NEIGHBORHOOD FILTERING (THE KEY PART!)
## ============================================

def filter_by_neighborhood(episodes, neighborhood):
    """
    Filter episodes for neighborhood relevance
    
    Strategy:
    - Look for neighborhood name OR major streets in that neighborhood
    - Use fuzzy matching (lowercase, strip punctuation)
    - Score each episode, keep ones above threshold
    """
    
    # Normalize neighborhood name
    neighborhood_normalized = neighborhood.lower().strip()
    
    # Get keyword list for this neighborhood
    keywords = NEIGHBORHOOD_KEYWORDS.get(neighborhood_normalized, [neighborhood_normalized])
    
    relevant_episodes = []
    
    for episode in episodes:
        # Combine title and description for searching
        text = f"{episode['title']} {episode['description']}".lower()
        
        # Check if any keyword appears
        score = 0
        matched_keywords = []
        
        for keyword in keywords:
            if keyword in text:
                score += 10
                matched_keywords.append(keyword)
        
        # Bonus points for contextual keywords
        context_keywords = [
            'development', 'construction', 'building', 'project',
            'opening', 'closing', 'new', 'restaurant', 'bar',
            'zoning', 'approved', 'proposed', 'retail',
            'apartment', 'condo', 'commercial', 'deal'
        ]
        
        for context in context_keywords:
            if context in text:
                score += 2
        
        # Include if score is high enough
        if score >= 10:  # At least one neighborhood keyword
            episode['relevance_score'] = score
            episode['matched_keywords'] = matched_keywords
            relevant_episodes.append(episode)
    
    # Sort by relevance score (highest first)
    relevant_episodes.sort(key=lambda x: x['relevance_score'], reverse=True)
    
    return relevant_episodes

def generate_podcast_section_for_report(neighborhood, property_type='residential'):
    """
    Generate the complete podcast section for property report
    """
    
    # Get podcast content
    content = get_podcast_content_for_property(neighborhood, property_type, days_back=60)
    
    # If no content found
    if not content['hospitality'] and not content['real_estate']:
        return f"""
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
RECENT PODCAST COVERAGE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

No recent podcast coverage found for {neighborhood} in the past 60 days.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
"""
    
    # Build report section
    report = f"""
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
RECENT PODCAST COVERAGE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Recent podcast episodes covering {neighborhood}:

"""
    
    # Add Real Estate episodes first (for most property types)
    if content['real_estate']:
        report += "\n🏗️ REAL ESTATE & DEVELOPMENT\n\n"
        
        for episode in content['real_estate'][:5]:  # Max 5 episodes
            report += format_episode(episode)
    
    # Add Hospitality episodes
    if content['hospitality']:
        report += "\n🍴 HOSPITALITY & CULTURE\n\n"
        
        for episode in content['hospitality'][:3]:  # Max 3 episodes
            report += format_episode(episode)
    
    report += """
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
"""
    
    return report


def format_episode(episode):
    """
    Format a single episode for display
    """
    # Truncate description
    description = episode['description']
    if len(description) > 200:
        description = description[:200] + "..."
    
    return f"""
📻 "{episode['title']}"
   {episode['podcast_name']} • {episode['published']}
   
   {description}
   
   🎧 Listen ({episode['duration']}): {episode['link']}

"""

        'neighborhood': 'Wicker Park',

    return podcast_section
